fun xaynha(build, trung){
	var lv = build[1]
	var time = build[2]
	
	
	while (!Region(1459, 620, 20, 20).find("build")) {
		click(Point(1600, 0))
		wait(delay)
	}
	wait(200)
	click(Point(1474, 641))
	var count = townLv
	wait(400)
	var targetBuild = Region(1130,55,70,575).findMulti(build[3], FParam.mScore(0.85))
	while(!targetBuild){
		if(count > 0){
			count = count - 1
			wait(25)
			swipe([SwipePoint(Point(1350, 290),10 , 40), SwipePoint(Point(1350, 120), 0, 40)], SParam.repeat(1).delay(0).waitNext(10).random(0))
		}else{
			count = townLv
			click(Point(1600, 0))
			while (!Region(1459, 620, 20, 20).find("build")) {
				click(Point(1600, 0))
				wait(delay)
			}
			wait(200)
			click(Point(1474, 641))
		}
		wait(400)
		targetBuild = Region(1130,55,70,575).findMulti(build[3], FParam.mScore(0.85))
		if(trung > 0 and targetBuild) {
			targetBuild = false
			trung = trung - 1
		}
	}
	
	click(targetBuild[0].getPoint())
	wait(200)
	
	if(lv > 1){ 
		if(!Region(600, 30, 600, 38).find(lv + "", FParam.mScore(0.9))) return xaynha(build, trung + 1)
		while (!Region(1004,583,30,30).find("confirm")) {wait(delay)}
		click(Point(1033, 624))
		if(build[0] == "townHall"){
			wait(200)
			if(Region(839, 596, 30, 30).find("continue")) click(Point(917, 625))
		}
	}else{
		if(build[0] == "clanCastle"){
			while (!Region(995,581,30,30).find("clanCastle")) {wait(delay)}
			click(Point(1033, 624))
		}else{
			var okBuild = Region(483, 60, 724, 618).find("ok")
			while (!okBuild) {
				wait(delay)
				okBuild = Region(483, 60, 724, 618).find("ok")
			}
			click(okBuild.getPoint())
		}
	}
	
	wait(200)
	click(Point(812, 464))
	setTimeBuild(build[0], time, lv)
	//ketthucbanggem()
	ketthucthuong()
	if(build[0] == "townHall"){
		wait(200)
		click(Point(1600, 0))
		wait(200)
		zoomMin.go()
	}
	
	//
}

fun buildUpgrade(){
	if (checkBuilderBusy() == "Bận") return "Bận"
	else {
		var build = buildSelect()
		xaynha(build, 0)
		copyAllData()
		if(build[0] == "townHall") Sys.pause()
		setUp()
		return "Không bận"
	}

}
