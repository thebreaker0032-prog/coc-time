fun xaynha(build){
	var lv = build[1]
	var time = build[2]
	
	while (!Region(1459, 620, 20, 20).find("build")) {
		click(Point(1600, 0))
		wait(delay)
	}
	wait(200)
	click(Point(1474, 641))
	wait(300)
	var targetBuild = Region(1130,55,70,575).find(build[3])
	while(!targetBuild){
		wait(delay)
		swipe([SwipePoint(Point(1350, 250), 10, 40), SwipePoint(Point(1350, 120), 0, 40)], SParam.repeat(1).delay(0).waitNext(10).random(0))
		wait(300)
		targetBuild = Region(1130,55,70,575).find(build[3], FParam.mScore(0.9))
	}
	click(targetBuild.getPoint())
	build = build[0]
	if(lv > 1){ 
		while (!Region(995,581,30,30).find("xacnhan")) {wait(delay)}
		click(Point(1033, 624))
	}else{
		if(build == "clanCastle"){
			while (!Region(995,581,30,30).find("clanCastle")) {wait(delay)}
			click(Point(1033, 624))
		}else{
			var okBuild = Region(483, 60, 724, 618).find("ok")
			while (!okBuild) {
				wait(delay)
				okBuild = Region(483, 60, 724, 618).find("ok")
			}
			click(okBuild.getPoint())
		}
	}
	//
	//setTimeBuild(build, Sys.currentTime() + time * 1000, lv)
}
