fun setAllBuildCompleted(){
	for (var i = 0; i < buildData.size; i = i + 1) {
		buildData[i][1] = Num.parse(buildData[i][1])
		if (buildData[i].size > 2 and Sys.currentTime() >= buildData[i][1]) {
			setUpgBuild(buildData[i][0], buildData[i][2])
			buildData[i] = ["", 0]
		}

	}
}

fun getBuildNext(buildInCheck){
	var build = a[id].get("orderBuild")[townLv]
	if(build.size == 0){
		build = a[id].get("buildLate").shift()
		buildInCheck[1].push(build)
		build = build.split("-")
		build = getInfoBuild(build[0], Num.parse(build[1]))
		return [build, buildInCheck]
	}else{
		build = build.shift()
		buildInCheck[0].push(build)
		build = build.split("-")
		build = getInfoBuild(build[0], Num.parse(build[1]))
		return [build, buildInCheck]
	}
}

fun checkBuildNext(build, buildInCheck){
	if(build[5] == "thiếu tài nguyên"){
		if(a[id].get(build[4]).size == 0) {
			build = getBuildNext(buildInCheck)
			return checkBuildNext(build[0], build[1])
		}else{
			build = getInfoBuild(build[4], Num.parse(a[id].get(build[4])[0]) + 1)
			return checkBuildNext(build, buildInCheck)
		}
	}else if(build[5] == "đang them tài nguyên") {
		build = getBuildNext(buildInCheck)
		return checkBuildNext(build[0], build[1])
	}else if(build[5] == "không thể xây"){
		for (var i = 0; i < 2; i = i + 1){
			var index = buildInCheck[i].indexOf(build[3])
			if (index > -1) {
				buildInCheck[i].pop()
				build = getBuildNext(buildInCheck)
				return checkBuildNext(build[0], build[1])
			}
		}
	}
	return build
}
