fun sortTime(a, b) {
	if (a == null or b == null) return 0
	a = Num.parse(a[1])
	b = Num.parse(b[1])
	if (a < b) return -1
	if (a > b) return 1
	return 0
}

fun getListTime(){
	var sorted = []
	
	for (var i = 0; i < a.size; i = i + 1) {
		var timeBuild = a[i].get("time")
		for (var j = 0; j < timeBuild.size; j = j + 1) {
			var item
			if(timeBuild[j].size > 2) item = [i, timeBuild[j][1], timeBuild[j][0] + "-" + timeBuild[j][2]]
			else item = [i, timeBuild[j][1], timeBuild[j][0]]
			sorted.push(item)
		}
	}
	sorted.sort(sortTime)
	var result = []
	var id = []
	for (var k = 0; k < sorted.size; k = k + 1) {
		var timeRemain = Num.parse(sorted[k][1]) - Sys.currentTime()
		result.push((sorted[k][0] + 1) + "-" + sorted[k][2] + ": " + Math.floor(timeRemain / (1000 * 60 * 60 * 24)) + "d" + Math.floor((timeRemain % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)) + "h" + Math.floor((timeRemain % (1000 * 60 * 60)) / (1000 * 60)) + "p" + Math.floor((timeRemain % (1000 * 60)) / 1000))
		id.push(sorted[k][0])
	}
	Sys.log(result)
	return id
}
fun upgradeStorage(buildName){
	var n = ["goldStorage", "elixirStorage", "darkElixirStorage"]
	n = n.indexOf(buildName)
	if (n > -1) {
		var town = buildList.get("townHall")
		if (n == 0) a[id].put("g", getStorage(buildName, town[0][townLv]))
		else if (n == 1) a[id].put("e", getStorage(buildName, town[0][townLv]))
		else a[id].put("d", getStorage(buildName, town[1][townLv]))
	}
}
fun upgradedTown(){

fun getMin(arr, max){
	if (arr == null or arr.size == 0) return -1
	var minItem = arr[0]
	for (var i = 1; i < arr.size; i = i + 1)
		if (minItem > arr[i]) minItem = arr[i]
	if (minItem == max) return 0
	Sys.log("get level từ " + arr + " : " + (minItem + 1))
	return minItem + 1
}
